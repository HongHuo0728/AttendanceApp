cmake_minimum_required(VERSION 3.20)
project(AttendanceApp VERSION 2.0.4 LANGUAGES CXX)
enable_language(RC)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB_RECURSE ATTENDANCE_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM ATTENDANCE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(AttendanceRuntime STATIC ${ATTENDANCE_SOURCES})
set_target_properties(AttendanceRuntime PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_executable(AttendanceApp WIN32 src/main.cpp resources/app.rc)

target_include_directories(AttendanceRuntime PUBLIC include src)
target_compile_definitions(AttendanceRuntime PUBLIC UNICODE _UNICODE)
if (MSVC)
    target_compile_options(AttendanceRuntime PUBLIC /utf-8 /W4 /Gy /Gw $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/Os>)
    set_property(TARGET AttendanceRuntime AttendanceApp PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    target_link_options(AttendanceApp PRIVATE /MANIFEST:NO /LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO)
endif()

target_link_libraries(AttendanceRuntime PUBLIC user32 gdi32 comdlg32 comctl32 shell32 ole32 uxtheme dwmapi msimg32)
target_link_libraries(AttendanceApp PRIVATE AttendanceRuntime)

include(CTest)
if (BUILD_TESTING)
    add_executable(AttendanceTests tests/regression.cpp)
    set_target_properties(AttendanceTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
    target_link_libraries(AttendanceTests PRIVATE AttendanceRuntime)
    if (MSVC)
        target_link_options(AttendanceTests PRIVATE /LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO)
    endif()
    add_test(NAME AttendanceRegression COMMAND AttendanceTests "${CMAKE_CURRENT_BINARY_DIR}/test-data")
    if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/baseline-2.0.3-ultra/src/main.cpp")
        add_executable(AttendanceBaseline EXCLUDE_FROM_ALL tests/regression.cpp)
        set_target_properties(AttendanceBaseline PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
        target_compile_definitions(AttendanceBaseline PRIVATE ATTENDANCE_BASELINE UNICODE _UNICODE)
        target_compile_options(AttendanceBaseline PRIVATE /utf-8 /O2 /Os /Gy /Gw /GL)
        target_link_options(AttendanceBaseline PRIVATE /LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO)
        target_link_libraries(AttendanceBaseline PRIVATE user32 gdi32 comdlg32 comctl32 shell32 ole32 uxtheme dwmapi msimg32)
    endif()
endif()
