cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ATTENDANCE_VERSION LIMIT_COUNT 1)
project(AttendanceApp VERSION ${ATTENDANCE_VERSION} LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ATTENDANCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
string(REPLACE "." "," ATTENDANCE_VERSION_COMMAS "${ATTENDANCE_VERSION}")
set(ATTENDANCE_MANIFEST_PATH "${CMAKE_BINARY_DIR}/generated/app.manifest")
configure_file(include/version.h.in generated/version.h @ONLY)
configure_file(resources/app.manifest.in generated/app.manifest @ONLY)
configure_file(resources/app.rc.in generated/app.rc @ONLY)
file(GLOB_RECURSE ATTENDANCE_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM ATTENDANCE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(AttendanceRuntime OBJECT ${ATTENDANCE_SOURCES})
if(CMAKE_GENERATOR MATCHES "Visual Studio")
 set_target_properties(AttendanceRuntime PROPERTIES VS_GLOBAL_ForceImportAfterCppTargets "${CMAKE_CURRENT_SOURCE_DIR}/scripts/object_only.targets")
endif()
add_executable(AttendanceApp WIN32 src/main.cpp "${CMAKE_BINARY_DIR}/generated/app.rc")
target_include_directories(AttendanceRuntime PUBLIC include src "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(AttendanceRuntime PUBLIC UNICODE _UNICODE)
if(MSVC)
 target_compile_options(AttendanceRuntime PUBLIC /utf-8 /W4 /Gy /Gw $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/Os>)
 set_property(TARGET AttendanceRuntime AttendanceApp PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
 target_link_options(AttendanceApp PRIVATE /MANIFEST:NO /LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO)
endif()
target_link_libraries(AttendanceRuntime PUBLIC user32 gdi32 comdlg32 comctl32 shell32 ole32 oleacc uxtheme dwmapi msimg32)
target_link_libraries(AttendanceApp PRIVATE AttendanceRuntime)
option(BUILD_TESTING "Build small isolated regression tests" OFF)
include(CTest)
if(BUILD_TESTING)
 add_executable(AttendanceTests tests/regression.cpp "${CMAKE_BINARY_DIR}/generated/app.rc")
 set_target_properties(AttendanceTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
 target_link_libraries(AttendanceTests PRIVATE AttendanceRuntime)
 if(MSVC)
  target_link_options(AttendanceTests PRIVATE /MANIFEST:NO /LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO)
 endif()
 add_test(NAME AttendanceRegression COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/run_tests.ps1" -SkipBuild -Runs 1)
endif()
